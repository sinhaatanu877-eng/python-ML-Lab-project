'''Question: Write a script to add a new dynamically calculated column named Grade to the DataFrame based on the Marks column (e.g., 90 is 'A', 80–89 is 'B', etc.).'''




import pandas as pd

# Create DataFrame
data = {
    "Student_Name": ["Amit", "Riya", "Sourav", "Neha", "Rahul"],
    "Roll_Number": [101, 102, 103, 104, 105],
    "Marks": [72, 85, 91, 68, 77],
    "Attendance": [88, 92, 76, 95, 81]
}

df = pd.DataFrame(data)

# Function to calculate Grade
def calculate_grade(marks):
    if marks >= 90:
        return "A"
    elif marks >= 80:
        return "B"
    elif marks >= 70:
        return "C"
    elif marks >= 60:
        return "D"
    else:
        return "F"

# Add Grade column
df["Grade"] = df["Marks"].apply(calculate_grade)

# Display DataFrame
print(df)
